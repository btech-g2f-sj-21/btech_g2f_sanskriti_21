#include <stdio.h>
#include <math.h>
int main() {
    float p,r,t,n;
    printf("enter p: ");
    scanf("%f",&p);
    printf("enter r: ");
    scanf("%f",&r);
    printf("enter t: ");
    scanf("%f",&t);
    printf("enter n: ");
    scanf("%f",&n);
printf("simple interest: %f",(p*r*t)/100);
printf("compound interest: %f",p*pow(1+(r/100),t)-p);
 
 return 0;
 

}
